Renderizado Gráfico vs. Simulación Física
Marco Teórico 1: Arquitectura de Motores y Sincronización
Cannon-es · simulación
La física (Cannon-es) corre en pasos de tiempo fijos 
(Fixed Timestep), independientemente de la tasa de 
refresco del monitor.
Calcula fuerzas, momentos de inercia y colisiones 
entre cuerpos rígidos (Body).
Three.js · renderizado
Bucle asíncrono: Three.js renderiza según la tasa de 
refresco del monitor.
La escena visual proyecta geometrías (Mesh).
Body
Cuerpo rígido calculado por la física.
Transferencia
Copia vectores de posición y rotaciones 
cuaterniónicas en cada fotograma.
Mesh
Representación gráfica sincronizada con 
el cuerpo físico.

Optimización de Colisiones en Entornos Masivos
Marco Teórico 2: Algoritmos de Colisión y Optimización
1
Broadphase · fase amplia
Descarta rápidamente objetos lejanos mediante 
volúmenes envolventes (AABB), con métodos como 
Sweep and Prune (SAP).
2
Narrowphase · fase estrecha
Calcula con precisión puntos de contacto e intersección geométrica 
solo entre entidades cercanas.
3 Estado «Sleep»
Congela cálculos numéricos en cuerpos cuya velocidad cae bajo un umbral y 
libera recursos de CPU.

// 2. MUNDO FÍSICO Y SUELO
const world = new CANNON.World({ gravity: new CANNON.Vec3(0, -9.81, 0) });
const groundBody = new CANNON.Body({ mass: 0, shape: new CANNON.Plane() });
groundBody.quaternion.setFromEuler(-Math.PI / 2, 0, 0);
world.addBody(groundBody);
const groundMesh = new THREE.Mesh(
    new THREE.PlaneGeometry(60, 60),
    new THREE.MeshStandardMaterial({ color: 0x333333 })
);
groundMesh.rotation.x = -Math.PI / 2;
scene.add(groundMesh);
// 3. CAJAS DINÁMICAS
const objects = [];
for (let i = 0; i < 20; i++) {
    const body = new CANNON.Body({ mass: 1, shape: new CANNON.Box(new CANNON.Vec3(0.5, 0.5, 0.5)) });
    body.position.set((Math.random() - 0.5) * 6, 2 + i * 1.2, (Math.random() - 0.5) * 6);
    world.addBody(body);
    const mesh = new THREE.Mesh(new THREE.BoxGeometry(1, 1, 1), new THREE.MeshStandardMaterial({ color: 0x00ff88 }));
    scene.add(mesh);
    objects.push({ body, mesh });
}

// 5. EVENTOS DE TECLADO
const keys = { w: false, a: false, s: false, d: false, space: false };
let canJump = false;
window.addEventListener('keydown', (e) => {
    if (e.key.toLowerCase() === 'w') keys.w = true;
    if (e.key.toLowerCase() === 'a') keys.a = true;
    if (e.key.toLowerCase() === 's') keys.s = true;
    if (e.key.toLowerCase() === 'd') keys.d = true;
    if (e.code === 'Space' && canJump) {
        playerBody.applyImpulse(new CANNON.Vec3(0, 25, 0), playerBody.position);
        canJump = false;
    }
});
window.addEventListener('keyup', (e) => {
    if (e.key.toLowerCase() === 'w') keys.w = false;
    if (e.key.toLowerCase() === 'a') keys.a = false;
    if (e.key.toLowerCase() === 's') keys.s = false;
    if (e.key.toLowerCase() === 'd') keys.d = false;
});

// 6. COLISIONES EN TIEMPO REAL
playerBody.addEventListener('collide', (event) => {
    const contact = event.contact;
    if (contact.bi.id === playerBody.id && contact.ni.y > 0.5) {
        canJump = true;
    } else if (contact.bj.id === playerBody.id && contact.ni.y < -0.5) {
        canJump = true;
    }
    if (event.contact.getImpactVelocityAlongNormal() > 3) {
        playerMesh.material.color.setHex(0xffff00);
        setTimeout(() => playerMesh.material.color.setHex(0xff0055), 150);
    }
});

// BONUS: MECÁNICA DE DISPARO (Instanciación dinámica)
window.addEventListener('click', () => {
    // 1. Obtener dirección hacia donde mira la cámara
    const shootDirection = new THREE.Vector3();
    camera.getWorldDirection(shootDirection);
    const projRadius = 0.3;
    // 2. Crear física del proyectil
    const projBody = new CANNON.Body({ mass: 2, shape: new CANNON.Sphere(projRadius) });
    projBody.position.copy(playerBody.position);
    projBody.position.y += 1; // Disparar un poco por encima del jugador
    // 3. Aplicar velocidad inicial agresiva
    const shootVelocity = 35;
    projBody.velocity.set(
        shootDirection.x * shootVelocity,
        shootDirection.y * shootVelocity,
        shootDirection.z * shootVelocity
    );
    world.addBody(projBody);

// 7. BUCLE DE ANIMACIÓN
function animate() {
    requestAnimationFrame(animate);
    world.fixedStep();
    const speed = 8;
    if (keys.w) playerBody.velocity.z = -speed;
    if (keys.s) playerBody.velocity.z = speed;
    if (keys.a) playerBody.velocity.x = -speed;
    if (keys.d) playerBody.velocity.x = speed;
    if (!keys.w && !keys.s) playerBody.velocity.z *= 0.9;
    if (!keys.a && !keys.d) playerBody.velocity.x *= 0.9;
    playerMesh.position.copy(playerBody.position);
    playerMesh.quaternion.copy(playerBody.quaternion);
    groundMesh.position.copy(groundBody.position);
    groundMesh.quaternion.copy(groundBody.quaternion);
    // Aquí se animan automáticamente tanto las cajas como los proyectiles
    objects.forEach(obj => {
        obj.mesh.position.copy(obj.body.position);
        obj.mesh.quaternion.copy(obj.body.quaternion);
    });
