UNIVERSIDAD COOPERATIVA DE COLOMBIA · PASO 1 DE 4
1. Importación e inicialización del mundo físico
import * as CANNON from 'cannon-es';
// Crear el mundo físico
const world = new CANNON.World({
    gravity: new CANNON.Vec3(0, -9.81, 0) // Gravedad terrestre en Y
});
world.broadphase = new CANNON.NaiveBroadphase(); // Detección de colisiones simple
// Materiales físicos (Fricción y Restitución/Rebote)
const materialFisicoBase = new CANNON.Material('base');
const contactoMaterial = new CANNON.ContactMaterial(
    materialFisicoBase,
    materialFisicoBase,
    { friction: 0.3, restitution: 0.7 } // 0.7 = Rebote alto (tipo pelota)
);
world.addContactMaterial(contactoMaterial);

estático
sueloBody.addShape(sueloShape);
sueloBody.quaternion.setFromEuler(-Math.PI / 2, 0, 0); // Rotar para que quede horizontal
world.addBody(sueloBody);

3. Objeto dinámico interactivo (pelota cayendo)
// A. PARTE VISUAL (Three.js)
const pelotaGeo = new THREE.SphereGeometry(0.5, 32, 32);
const pelotaMat = new THREE.MeshStandardMaterial({ color: 0xff0055, roughness: 0.2 });
const pelotaMesh = new THREE.Mesh(pelotaGeo, pelotaMat);
scene.add(pelotaMesh);
// B. PARTE FÍSICA (Cannon.js)
const pelotaShape = new CANNON.Sphere(0.5);
const pelotaBody = new CANNON.Body({
    mass: 1, // Masa > 0 = Objeto afectado por la gravedad
    position: new CANNON.Vec3(0, 10, 0), // Nace a 10 metros de altura
    shape: pelotaShape,
    material: materialFisicoBase
});
world.addBody(pelotaBody);

4. Sincronización dentro del bucle animate()
const clock = new THREE.Clock();
let tiempoAnterior = 0;
function animate() {
    requestAnimationFrame(animate);
    const tiempoTranscurrido = clock.getElapsedTime();
    const deltaTime = tiempoTranscurrido - tiempoAnterior;
    tiempoAnterior = tiempoTranscurrido;
    // 1. Avanzar la simulación física (paso de tiempo fijo 60fps)
    world.step(1 / 60, deltaTime, 3);
    // 2. Sincronizar la malla de Three.js con el cuerpo físico de Cannon.js
    pelotaMesh.position.copy(pelotaBody.position);
    pelotaMesh.quaternion.copy(pelotaBody.quaternion);
    // ... (resto de animaciones de portal, partículas y triggers) ...
    renderer.render(scene, camera);
}
